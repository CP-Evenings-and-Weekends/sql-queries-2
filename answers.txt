2. Get the count of users per city, excluding Chicago.
   117 rows.

   SELECT COUNT(*), city FROM UserProfile WHERE NOT city = 'Chicago' GROUP BY city ORDER BY COUNT(*) DESC;

------------------------------------------------------------------------

3. Get the number of advertisements per date in descending order.
   330 rows.

   SELECT COUNT(*), advertisement_date FROM Advertisement GROUP BY advertisement_date ORDER BY COUNT(*) DESC;

4. Get the number of advertisements per date before February 2014, in descending order.
   23 rows.

   SELECT COUNT(*), advertisement_date FROM Advertisement WHERE advertisement_date < '2014-02-01' GROUP BY advertisement_date ORDER BY COUNT(*) DESC;

6. What are the first names of users who placed ads in June of 2015?
   0 rows.

   SELECT AppUser.first_name
   FROM AppUser 
   INNER JOIN Advertisement 
   ON AppUser.account_id = Advertisement.seller_account_id
   WHERE extract(MONTH FROM Advertisement.advertisement_date) = 6
   AND extract(YEAR FROM Advertisement.advertisement_date) = 2015;

7. What are the makes of cars that Wilda Giguere advertised?
       make    
   ------------
    BMW
    Opel
    Peugeot
    Volkswagen
   4 rows.

   SELECT DISTINCT CarModel.make
   FROM AppUser
   JOIN Advertisement ON AppUser.account_id = Advertisement.seller_account_id
   JOIN Car ON Advertisement.car_id = Car.car_id
   JOIN CarModel ON Car.car_model_id = CarModel.car_model_id
   WHERE AppUser.first_name = 'Wilda' AND AppUser.last_name = 'Giguere';
