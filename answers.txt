Get the count of users per city.

SELECT city, COUNT(*) AS user_count
FROM UserProfile
GROUP BY city;


Get the count of users per city, excluding Chicago.


SELECT city, COUNT(*) AS user_count
FROM UserProfile
WHERE city <> 'Chicago'
GROUP BY city;

Get the number of advertisements per date in descending order.

SELECT advertisement_date::date AS ad_date, COUNT(*) AS ad_count
FROM advertisement
GROUP BY advertisement_date::date
ORDER BY ad_date DESC;

Get the number of advertisements per date before February 2014, in descending order.

SELECT advertisement_date::date AS ad_date, COUNT(*) AS ad_count
FROM advertisement
WHERE advertisement_date < '2014-02-01'
GROUP BY advertisement_date::date
ORDER BY ad_date DESC;


Get the number of advertisements per month.

SELECT DATE_TRUNC('month', advertisement_date) AS ad_month, COUNT(*) AS ad_count
FROM advertisement
GROUP BY DATE_TRUNC('month', advertisement_date)
ORDER BY ad_month;


What are the first names of users who placed ads in June of 2015?

SELECT DISTINCT au.first_name
FROM appuser au
JOIN advertisement a ON au.account_id = a.seller_account_id
WHERE a.advertisement_date >= '2015-06-01' AND a.advertisement_date < '2015-07-01';



What are the makes of cars that Wilda Giguere advertised?

SELECT DISTINCT cm.make
FROM carmodel cm
JOIN car c ON cm.car_model_id = c.car_model_id
JOIN advertisement a ON c.car_id = a.car_id
JOIN appuser au ON au.account_id = a.seller_account_id
WHERE au.first_name = 'Wilda' AND au.last_name = 'Giguere';
